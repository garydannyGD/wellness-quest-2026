-- WELLNESS QUEST 2026 -- Run in the Supabase SQL Editor. create table if not exists public.scores ( id bigint generated always as identity primary key, player_id uuid not null, nickname text not null check ( nickname ~ '^[A-Za-z0-9 _-]{2,18}$' ), game text not null check (game in ('quiz','life')), score integer not null check (score between 0 and 100), created_at timestamptz not null default now() ); create index if not exists scores_player_idx on public.scores (player_id, game); create index if not exists scores_created_idx on public.scores (created_at desc); -- Apply row-level security. alter table public.scores enable row level security; -- Remove broad public permissions. revoke all on public.scores from anon, authenticated; -- Permit only public reading and score submission. grant select, insert on public.scores to anon; -- Reading policy. drop policy if exists "Public score reading" on public.scores; create policy "Public score reading" on public.scores for select to anon using (true); -- Submission policy. drop policy if exists "Public score submission" on public.scores; create policy "Public score submission" on public.scores for insert to anon with check ( score between 0 and 100 and game in ('quiz','life') and nickname ~ '^[A-Za-z0-9 _-]{2,18}$' ); -- Aggregate each browser's highest score -- from each game. create or replace view public.leaderboard with (security_invoker = true) as with best_scores as ( select player_id, max(nickname) as nickname, max(score) filter ( where game = 'quiz' ) as quiz_score, max(score) filter ( where game = 'life' ) as life_score, count(distinct game) as games_played from public.scores group by player_id ) select player_id, nickname, coalesce(quiz_score,0)::integer as quiz_score, coalesce(life_score,0)::integer as life_score, ( coalesce(quiz_score,0) + coalesce(life_score,0) )::integer as total, games_played from best_scores; revoke all on public.leaderboard from anon, authenticated; grant select on public.leaderboard to anon;
